@echo off
chcp 65001 >nul
setlocal
cd /d "%~dp0"
rem SPRITE-ONLY bake-off (2026-11-10): Sarah's game sprite on every model x LoRA x sampler, each sampler at its own
rem optimal steps/cfg/scheduler, 2 generations (seed changes each generation; same 2 seeds in every cell).
rem   SDXL:  5 checkpoints x 27 LoRAs (+ none) x 14 samplers x 2 seeds = ~3,220 renders, ~10-11 h on the 3080 Ti
rem   FLUX:  klein 4B base + distilled x 5 samplers, schnell x 6 samplers (commercial-OK only) = 32 renders
rem What it does: parks everything already queued (your current bake-off waits safely in jobs\paused\),
rem queues the sprite suites at top priority, and starts the worker in its own window.
rem Bring the old bake-off back afterwards with resume.bat.
if not exist venv (echo Run install.bat first. & pause & exit /b 1)
call venv\Scripts\activate.bat
echo.
echo  STEP 1 - stop the running worker first:
echo    click its window and press Ctrl+C (answer Y if it asks), then close it.
echo    The job it was on is not lost - it goes back in the queue below.
echo    (The worker MUST restart - it has to load the new samplers.)
echo.
pause
echo.
echo Returning any interrupted job to the queue ...
python -m forge requeue
echo Parking everything that was queued (resume.bat brings it back) ...
python -m forge pause
echo.
echo Scanning model folders (now includes models\diffusion_models for klein / schnell) ...
python -m forge models scan >nul
echo.
python -m forge bakeoff submit games\turing_accords\bakeoff\sprite_samplers_v1.json --rescan
python -m forge bakeoff submit games\turing_accords\bakeoff\sprite_samplers_klein_v1.json
python -m forge bakeoff submit games\turing_accords\bakeoff\sprite_samplers_schnell_v1.json
echo.
echo Starting the worker in its own window ...
start "Forge worker" "%~dp0run_worker.bat"
echo.
echo  Done. Results build up as it runs - look any time with:
echo    bakeoff_report.bat sprite_samplers_v1
echo    (across_sarah_*.png = rows models, columns samplers, one sheet per LoRA and seed)
echo    bakeoff_report.bat sprite_samplers_klein_v1   /   bakeoff_report.bat sprite_samplers_schnell_v1
echo  Progress / time left: eta.bat      When finished: resume.bat to restart the old bake-off.
echo.
pause
